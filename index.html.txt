  var tiemposRespuestaPQRS = {

  var textoTiempoRespuesta = document.getElementById('rad-tiempo-respuesta');
  function actualizarTiempoRespuesta(valorSeleccionado){
    var dias = tiemposRespuestaPQRS[valorSeleccionado];
    textoTiempoRespuesta.textContent = dias ? 'Tiempo de respuesta estimado: ' + dias : '';

  // ---- Identidad (Anónimo / Ciudadano / Persona natural / jurídica / Contratista / Funcionario) ----
  var selectIdentidad = document.getElementById('rad-identidad');

  var bloqueDatosPersonales = document.getElementById('rad-datos-personales');
  var bloqueDatosPersonales2 = document.getElementById('rad-datos-personales-2');
  var camposPersonales = ['rad-tipodoc', 'rad-numdoc', 'rad-nombre'];
  var bloqueContratista = document.getElementById('rad-datos-contratista');
  var campoNumContrato = document.getElementById('rad-numcontrato');

  var campoDependencia = document.getElementById('rad-dependencia');

  var fieldsetRegistros = document.getElementById('rad-registros-fieldset');

  var botonAgregarRegistro = document.getElementById('rad-agregar-registro');
  var contadorRegistros = 0;

  function crearFilaRegistro(){
    contadorRegistros++;
    var fila = document.createElement('div');
    fila.className = 'registro-item';
    fila.innerHTML =
      '<p class="registro-num">Registro ' + contadorRegistros + '</p>' +
      '<div class="grid2">' +
        '<div class="campo">' +
          '<label>Tipo *</label>' +
          '<select class="reg-tipo" required>' +
            '<option value="">Selecciona</option>' +
            '<option value="chip">Chip catastral</option>' +
            '<option value="placa">Placa</option>' +
          '</select>' +
        '</div>' +
        '<div class="campo">' +
          '<label>Número de chip o placa *</label>' +
          '<input class="reg-numero" type="text" required placeholder="Escribe el chip o la placa">' +
        '</div>' +
      '</div>' +
      '<div class="campo">' +
        '<label>Número de proceso (opcional)</label>' +
        '<input class="reg-proceso" type="text" placeholder="Si tienes un proceso asociado">' +
      '</div>' +
      '<button type="button" class="registro-eliminar">Eliminar este registro</button>';
    listaRegistros.appendChild(fila);
    fila.querySelector('.registro-eliminar').addEventListener('click', function(){
      fila.remove();
      renumerarRegistros();
    });
  }

  function renumerarRegistros(){
    var filas = listaRegistros.querySelectorAll('.registro-item');
    filas.forEach(function(fila, i){
      fila.querySelector('.registro-num').textContent = 'Registro ' + (i + 1);
    });
    if(filas.length === 0){ crearFilaRegistro(); }
  }

  botonAgregarRegistro.addEventListener('click', crearFilaRegistro);

  function filtrarTipoDocumento(perfil){
    var opciones = campoTipoDoc.querySelectorAll('option[data-perfiles]');
    opciones.forEach(function(opt){
      var perfiles = opt.getAttribute('data-perfiles').split(' ');
      opt.hidden = perfiles.indexOf(perfil) === -1;
    });
    var actual = campoTipoDoc.selectedOptions[0];
    if(actual && actual.hidden){
      campoTipoDoc.value = '';
    }
    actualizarEtiquetaNombre();
  }

  var campoNombre = document.getElementById('rad-nombre');
  var etiquetaNombre = document.getElementById('rad-nombre-label');
  function actualizarEtiquetaNombre(){
    var esNIT = campoTipoDoc.value === 'NIT';
    etiquetaNombre.textContent = esNIT ? 'Razón social *' : 'Nombres y apellidos *';
    campoNombre.placeholder = esNIT ? 'Razón social de la empresa' : 'Nombre completo';
  }
  campoTipoDoc.addEventListener('change', actualizarEtiquetaNombre);

  // ---- Trámite: selección, resumen y modal de descripción ----
  var selectTramite = document.getElementById('tr-tipo');
  var cajaResumen = document.getElementById('tr-resumen');
  var resumenTitulo = document.getElementById('tr-resumen-titulo');
  var resumenAlias = document.getElementById('tr-resumen-alias');
  var botonVerDescripcion = document.getElementById('tr-ver-descripcion');

  var modal = document.getElementById('tr-modal');
  var modalTitulo = document.getElementById('tr-modal-titulo');
  var modalAlias = document.getElementById('tr-modal-alias');
  var modalDescripcion = document.getElementById('tr-modal-descripcion');
  var modalEntidad = document.getElementById('tr-modal-entidad');

  function abrirModalTramite(val){
    var t = tramitesData[val];
    if(!t) return;
    modalTitulo.textContent = t.titulo;
    modalAlias.textContent = t.alias ? 'También se conoce como: ' + t.alias : '';
    modalAlias.style.display = t.alias ? 'block' : 'none';
    modalDescripcion.textContent = t.descripcion;
    modalEntidad.textContent = 'Entidad responsable: ' + t.entidad;
    modal.classList.add('is-visible');
  }
  function cerrarModalTramite(){
    modal.classList.remove('is-visible');
  }

  selectTramite.addEventListener('change', function(){
    var val = selectTramite.value;
    if(!val){ cajaResumen.style.display = 'none'; return; }
    var t = tramitesData[val];
    resumenTitulo.textContent = t.titulo;
    resumenAlias.textContent = t.alias ? 'También se conoce como: ' + t.alias : '';
    cajaResumen.style.display = 'block';
    abrirModalTramite(val);
  });

  botonVerDescripcion.addEventListener('click', function(){ abrirModalTramite(selectTramite.value); });
  document.getElementById('tr-modal-cerrar').addEventListener('click', cerrarModalTramite);
  modal.addEventListener('click', function(e){ if(e.target === modal) cerrarModalTramite(); });
  document.addEventListener('keydown', function(e){ if(e.key === 'Escape') cerrarModalTramite(); });

  function filtrarOpcionesTramite(perfil){
    var perfilMostrar = (perfil === 'contratista' || perfil === 'funcionario') ? perfil : 'general';
    var opciones = selectTramite.querySelectorAll('option[data-perfil]');
    opciones.forEach(function(opt){
      opt.hidden = (opt.getAttribute('data-perfil') !== perfilMostrar);
    });
    var actual = selectTramite.selectedOptions[0];
    if(actual && actual.hidden){
      selectTramite.value = '';
      cajaResumen.style.display = 'none';
      cerrarModalTramite();
    }
  }

  selectIdentidad.addEventListener('change', function(){
    var val = selectIdentidad.value;
    var esAnonimo = val === 'anonimo';
    var esContratista = val === 'contratista';
    var necesitaRegistros = (val === 'persona-natural' || val === 'persona-juridica');

    bloqueDatosPersonales.style.display = esAnonimo ? 'none' : 'grid';
    bloqueDatosPersonales2.style.display = esAnonimo ? 'none' : 'grid';
    camposPersonales.forEach(function(id){
      document.getElementById(id).required = !esAnonimo;
    });

    bloqueContratista.style.display = esContratista ? 'grid' : 'none';
    campoNumContrato.required = esContratista;
    campoTipoPersona.required = esContratista;
    campoDependencia.required = esContratista;

    fieldsetRegistros.style.display = necesitaRegistros ? 'block' : 'none';
    if(necesitaRegistros && listaRegistros.children.length === 0){
      crearFilaRegistro();
    }
    listaRegistros.querySelectorAll('.reg-tipo, .reg-numero').forEach(function(campo){
      campo.required = necesitaRegistros;
    });

    filtrarTipoDocumento(val);
    filtrarOpcionesTramite(val);
  });

  filtrarTipoDocumento('');
  filtrarOpcionesTramite('');

  // ---- Tipo de solicitud: PQRS vs Trámite ----
  var fieldsetTramite = document.getElementById('rad-tramite-fieldset');
  var fieldsetDescPqrs = document.getElementById('rad-pqrs-desc-fieldset');
  var fieldsetDocsTramite = document.getElementById('rad-tramite-docs-fieldset');
  var campoAsunto = document.getElementById('rad-asunto');
  var campoDescripcion = document.getElementById('rad-descripcion');
  var campoAdjuntoTramite = document.getElementById('rad-adjunto-tramite');
  var botonEnviar = document.getElementById('rad-submit-btn');

  function actualizarTipoSolicitud(valor){
    var esTramite = valor === 'tramite';

    fieldsetTramite.style.display = esTramite ? 'block' : 'none';
    selectTramite.required = esTramite;

    fieldsetDescPqrs.style.display = esTramite ? 'none' : 'block';
    campoAsunto.required = !esTramite;
    campoDescripcion.required = !esTramite;

    fieldsetDocsTramite.style.display = esTramite ? 'block' : 'none';
    campoAdjuntoTramite.required = esTramite;

    botonEnviar.textContent = esTramite ? 'Enviar solicitud de trámite' : 'Radicar solicitud';

    if(esTramite){
      textoTiempoRespuesta.textContent = '';
    } else {
      actualizarTiempoRespuesta(valor);
    }
  }

  document.querySelectorAll('input[name="tipo-solicitud"]').forEach(function(radio){
    radio.addEventListener('change', function(){ actualizarTipoSolicitud(radio.value); });
  });
  actualizarTipoSolicitud('peticion');

  // ---- Envío del formulario unificado ----
  function generarRadicado(prefijo){
    var anio = new Date().getFullYear();
    var num = Math.floor(100000 + Math.random() * 899999);
    return prefijo + '-' + anio + '-' + num;
  }

  var formRadicar = document.getElementById('form-radicar');
  var radError = document.getElementById('rad-error');
  var radConfirmacion = document.getElementById('rad-confirmacion');
  var radConfirmacionTitulo = document.getElementById('rad-confirmacion-titulo');
  var radConfirmacionTexto = document.getElementById('rad-confirmacion-texto');
  var radRadicado = document.getElementById('rad-radicado');

  formRadicar.addEventListener('submit', function(e){
    e.preventDefault();
    if(!formRadicar.checkValidity()){
      radError.style.display = 'block';
      formRadicar.reportValidity();
      return;
    }
    radError.style.display = 'none';
    var esTramite = document.querySelector('input[name="tipo-solicitud"]:checked').value === 'tramite';
    radConfirmacionTitulo.textContent = esTramite ? 'Tu trámite fue radicado' : 'Tu solicitud fue radicada';
    radConfirmacionTexto.textContent = esTramite
      ? 'Guarda este número: lo necesitas para hacer seguimiento a tu trámite en el portal o por la línea de atención.'
      : 'Guarda este número: lo necesitas para hacer seguimiento a tu PQRS en el portal o por la línea de atención.';
    radRadicado.textContent = generarRadicado(esTramite ? 'TRM' : 'PQRS');
    formRadicar.style.display = 'none';
    radConfirmacion.classList.add('is-visible');
  });

  // ---- Formulario de consulta de radicado ----
  var formConsultar = document.getElementById('form-consultar');
  var consError = document.getElementById('cons-error');
  var consResultado = document.getElementById('cons-resultado');
  var consTitulo = document.getElementById('cons-titulo');
  var consDetalle = document.getElementById('cons-detalle');
  var consRadicadoMostrado = document.getElementById('cons-radicado-mostrado');

  formConsultar.addEventListener('submit', function(e){
    e.preventDefault();
    var valor = document.getElementById('cons-radicado').value.trim().toUpperCase();
    var patron = /^(PQRS|TRM)-\d{4}-\d{5,6}$/;
    if(!patron.test(valor)){
      consError.style.display = 'block';
      formConsultar.reportValidity();
      return;
    }
    consError.style.display = 'none';
    var esPQRS = valor.indexOf('PQRS') === 0;
    consTitulo.textContent = esPQRS ? 'Tu PQRS está en trámite' : 'Tu trámite está en proceso';
    consDetalle.textContent = esPQRS
      ? 'Está siendo gestionada por la dependencia responsable dentro del tiempo de respuesta estipulado.'
      : 'Está siendo revisado por la dependencia responsable dentro del tiempo de respuesta estipulado.';
    consRadicadoMostrado.textContent = valor;
    formConsultar.style.display = 'none';
    consResultado.classList.add('is-visible');
  });

  // ---- Reiniciar formularios ----
  document.querySelectorAll('[data-reiniciar]').forEach(function(btn){
    btn.addEventListener('click', function(){
      var tipo = btn.getAttribute('data-reiniciar');
      if(tipo === 'consultar'){
        formConsultar.reset();
        formConsultar.style.display = 'block';
        consResultado.classList.remove('is-visible');
        return;
      }
      if(tipo === 'radicar'){
        formRadicar.reset();
        formRadicar.style.display = 'block';
        radConfirmacion.classList.remove('is-visible');
        listaRegistros.innerHTML = '';
        contadorRegistros = 0;
        selectIdentidad.dispatchEvent(new Event('change'));
        document.getElementById('tp-peticion').checked = true;
        actualizarTipoSolicitud('peticion');
        cajaResumen.style.display = 'none';
      }
    });
  });
})();
